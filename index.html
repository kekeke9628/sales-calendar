매출 캘린더 연결

Supabase 프로젝트 주소와 anon key를 넣으면 이 기기에 기억됩니다. anon key는 공개돼도 안전한 값이지만, 아래 SQL의 RLS 설정을 반드시 먼저 적용하세요.

Supabase에 먼저 실행할 SQL (필수) 
-- 1. 일자별 실적
create table daily_metrics (
  user_id      uuid not null default auth.uid() references auth.users,
  date         date not null,
  sales        bigint not null,    -- VAT 제외, 원 단위
  cars         bigint not null,
  target_sales bigint,
  note         text,
  updated_at   timestamptz default now(),
  primary key (user_id, date)
);
alter table daily_metrics enable row level security;
create policy "own" on daily_metrics for all
  using (auth.uid() = user_id) with check (auth.uid() = user_id);

-- 2. 앱 설정
create table app_settings (
  user_id uuid not null default auth.uid() references auth.users,
  key text not null, value jsonb,
  primary key (user_id, key)
);
alter table app_settings enable row level security;
create policy "own" on app_settings for all
  using (auth.uid() = user_id) with check (auth.uid() = user_id);

-- 3. 날씨 캐시 (민감정보 없음, 공용)
create table weather_cache (
  date date not null, lat numeric not null, lon numeric not null,
  payload jsonb not null,
  primary key (date, lat, lon)
);
alter table weather_cache enable row level security;
create policy "read" on weather_cache for select using (true);
create policy "ins"  on weather_cache for insert to authenticated with check (true);
create policy "upd"  on weather_cache for update to authenticated using (true) with check (true);

-- 4. 대비일 프로파일
create table compare_profiles (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null default auth.uid() references auth.users,
  name text not null,
  offset_days integer,
  created_at timestamptz default now()
);
alter table compare_profiles enable row level security;
create policy "own" on compare_profiles for all
  using (auth.uid() = user_id) with check (auth.uid() = user_id);

-- 5. 일자별 대비일 매핑
create table compare_mappings (
  profile_id uuid not null references compare_profiles(id) on delete cascade,
  user_id uuid not null default auth.uid() references auth.users,
  base_date date not null,
  compare_date date not null,
  primary key (profile_id, base_date)
);
alter table compare_mappings enable row level security;
create policy "own" on compare_mappings for all
  using (auth.uid() = user_id) with check (auth.uid() = user_id);
